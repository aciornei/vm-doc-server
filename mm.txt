1. Topul modificărilor din baza de date

SELECT
    category,
    object_type,
    change_type,
    COUNT(*) AS total_modificari,
    COUNT(DISTINCT vm_id) AS vm_afectate
FROM inventory_changes
WHERE occurred_at >= NOW() - INTERVAL 30 DAY
GROUP BY category, object_type, change_type
ORDER BY total_modificari DESC
LIMIT 30;

2. Care obiecte generează cel mai mult zgomot?

SELECT
    category,
    object_type,
    object_label,
    COUNT(*) AS modificari,
    COUNT(DISTINCT vm_id) AS vm_afectate
FROM inventory_changes
WHERE occurred_at >= NOW() - INTERVAL 30 DAY
GROUP BY category, object_type, object_label
ORDER BY modificari DESC
LIMIT 50;

3. Grupăm serviciile care își schimbă numărul instanței

SELECT
    REGEXP_REPLACE(
        object_key,
        '@[0-9]+',
        '@#'
    ) AS serviciu_normalizat,
    change_type,
    COUNT(*) AS total,
    COUNT(DISTINCT vm_id) AS vm_afectate
FROM inventory_changes
WHERE category = 'services'
  AND occurred_at >= NOW() - INTERVAL 30 DAY
GROUP BY serviciu_normalizat, change_type
ORDER BY total DESC
LIMIT 50;


SELECT
    COUNT(*) AS total,
    SUM(NOT (
        JSON_UNQUOTE(JSON_EXTRACT(old_json,'$.active_state'))
        <=>
        JSON_UNQUOTE(JSON_EXTRACT(new_json,'$.active_state'))
    )) AS active_state,
    SUM(NOT (
        JSON_UNQUOTE(JSON_EXTRACT(old_json,'$.sub_state'))
        <=>
        JSON_UNQUOTE(JSON_EXTRACT(new_json,'$.sub_state'))
    )) AS sub_state,
    SUM(NOT (
        JSON_UNQUOTE(JSON_EXTRACT(old_json,'$.enabled_state'))
        <=>
        JSON_UNQUOTE(JSON_EXTRACT(new_json,'$.enabled_state'))
    )) AS enabled_state,
    SUM(NOT (
        JSON_UNQUOTE(JSON_EXTRACT(old_json,'$.package_version'))
        <=>
        JSON_UNQUOTE(JSON_EXTRACT(new_json,'$.package_version'))
    )) AS package_version
FROM inventory_changes
WHERE category = 'services'
  AND change_type = 'modified'
  AND occurred_at >= NOW() - INTERVAL 30 DAY;

SELECT
  CASE
    WHEN CAST(SUBSTRING_INDEX(object_label, '/', -1)
              AS UNSIGNED) < 1024
      THEN '0-1023'
    WHEN CAST(SUBSTRING_INDEX(object_label, '/', -1)
              AS UNSIGNED) < 32768
      THEN '1024-32767'
    WHEN CAST(SUBSTRING_INDEX(object_label, '/', -1)
              AS UNSIGNED) < 49152
      THEN '32768-49151'
    ELSE '49152-65535'
  END AS interval_porturi,
  COUNT(*) AS total,
  SUM(change_type = 'added') AS adaugate,
  SUM(change_type = 'removed') AS eliminate,
  COUNT(DISTINCT vm_id) AS vm_afectate
FROM inventory_changes
WHERE category = 'ports'
  AND occurred_at >= NOW() - INTERVAL 30 DAY
GROUP BY interval_porturi
ORDER BY total DESC;


SELECT
  SUBSTRING_INDEX(
    SUBSTRING_INDEX(object_key, '|', 3),
    '|', -1
  ) AS backend,
  change_type,
  COUNT(*) AS total,
  COUNT(DISTINCT vm_id) AS vm_afectate
FROM inventory_changes
WHERE category = 'firewall'
  AND object_type = 'rule'
  AND occurred_at >= NOW() - INTERVAL 30 DAY
GROUP BY backend, change_type
ORDER BY total DESC;
